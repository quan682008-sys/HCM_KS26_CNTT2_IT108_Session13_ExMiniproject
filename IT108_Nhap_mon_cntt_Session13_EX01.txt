Vào powershell
mkdir greentech-systems
cd .\greentech-systems\
cd .\scripts\
New-Item auto_report.bat


Vào CMD
cd greentech-systems
cd scripts
.\auto_report.bat
echo @echo off >> auto_report.bat (để ẩn lệnh rác)
echo echo === Bao cao chuyen nghiep === > auto_report.bat (thông báo ra màn hình)
echo pause >> auto_report.bat (lệnh dùng để dừng màn hình khi mở file)
type log 500MB.log | findstr "CRITICAL", "ERROR"
